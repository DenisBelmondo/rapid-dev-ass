Implementing "Hold-to-Remove" Pylon Mechanic

Here is a step-by-step guide to changing the pylon removal mechanic from a simple press to a "hold-to-confirm" action, complete with visual feedback.

Step 1: Create the Visual Feedback UI in Unity

First, you need to create the radial progress bar. You'll do this in the Unity Editor.

  1. Open your Pylon prefab.
  2. Right-click the Pylon in the Hierarchy, go to UI -> Canvas, and add a new Canvas as a child of the Pylon. Let's name it HoldCanvas.
  3. Set the Render Mode of the HoldCanvas to World Space. This makes the UI exist in the game world, attached to the pylon. You'll need to drastically shrink its Rect Transform (e.g., set Width/Height to 1, and scale X/Y/Z to something like 0.01). Adjust it until it has a reasonable size relative to your pylon.
  4. Right-click HoldCanvas, go to UI -> Image, and add an Image. Name it ProgressRadial.
  5. Select ProgressRadial. In the Inspector for its Image component:
    • Assign a circular sprite to the Source Image (you can use the default "Knob" or any white circle).
    • Change Image Type to Filled.
    • Change Fill Method to Radial 360.
    • Set Fill Origin to Top.
    • By default, set Fill Amount to 0.
    • You can disable the ProgressRadial GameObject by default so it's not visible initially.

Step 2: Create PylonHoldFeedback.cs

Create a new C# script with this name and place it on your Pylon prefab. This script will control the ProgressRadial you just made.


using UnityEngine;
using UnityEngine.UI;

namespace Level
{
    // This script manages the visual feedback for holding to remove a pylon.
    // Place this on the Pylon prefab.
    public class PylonHoldFeedback : MonoBehaviour
    {
        [Tooltip("The radial Image component for showing hold progress.")]
        [SerializeField] private Image _progressRadial;

        private void Awake()
        {
            if (_progressRadial == null)
            {
                Debug.LogError("ProgressRadial is not assigned in PylonHoldFeedback.", this);
                return;
            }
            // Ensure it's hidden at the start.
            _progressRadial.gameObject.SetActive(false);
            _progressRadial.fillAmount = 0;
        }

        // Call this to begin showing the hold progress.
        public void StartHold()
        {
            _progressRadial.gameObject.SetActive(true);
            _progressRadial.fillAmount = 0;
        }

        // Updates the fill amount of the radial image.
        public void UpdateProgress(float progress)
        {
            if (!_progressRadial.gameObject.activeInHierarchy)
            {
                _progressRadial.gameObject.SetActive(true);
            }
            _progressRadial.fillAmount = Mathf.Clamp01(progress);
        }

        // Call this to cancel or complete the hold visual.
        public void StopHold()
        {
            _progressRadial.fillAmount = 0;
            _progressRadial.gameObject.SetActive(false);
        }
    }
}

After creating this script, make sure to drag the ProgressRadial Image GameObject from your prefab's hierarchy into the _progressRadial slot in the Inspector for PylonHoldFeedback.

Step 3: Modify Existing Scripts

Now, let's update Pylon.cs, PlayerInputController.cs, and PylonManager.cs to handle the new hold logic.

Pylon.cs (Modified)

This script now gets a reference to the PylonHoldFeedback and provides public methods to control it.


using Managers;
using TMPro;
using UnityEngine;

namespace Level
{
    public class Pylon : MonoBehaviour
    {
        private PylonManager _pylonManager;
        [SerializeField] private TMP_Text _hintText;
        private PylonHoldFeedback _holdFeedback; // NEW: Reference to the feedback script.

        private void Awake()
        {
            _pylonManager = World.Instance.pylonManager;
            _holdFeedback = GetComponent<PylonHoldFeedback>(); // NEW: Get the component.
            if (_hintText != null) _hintText.text = "";
        }

        // NEW: Public method for the controller to update this pylon's visuals.
        public void UpdateHoldProgress(float progress)
        {
            if (_holdFeedback == null) return;
            
            if (progress > 0)
            {
                _holdFeedback.UpdateProgress(progress);
            }
            else
            {
                _holdFeedback.StopHold();
            }
        }
        
        // NEW: Public method to cancel the hold visuals if the player walks away.
        public void CancelHold()
        {
            if (_holdFeedback != null)
            {
                _holdFeedback.StopHold();
            }
        }

        private void OnTriggerEnter2D(Collider2D other)
        {
            if (!other.CompareTag("Player")) return;
            
            _pylonManager.OnPylonInteracted(this.gameObject);

            bool isTriangleCompletionPylon =
                _pylonManager.activePylons.Count == 3 && _pylonManager.activePylons[0] == gameObject;

            if (isTriangleCompletionPylon) return;
            
            Debug.Log("Can be removed!");
            _pylonManager.pylonToRemove = this.gameObject;
            if (_hintText != null) _hintText.text = "Hold Space to Remove";
        }

        private void OnTriggerExit2D(Collider2D other)
        {
            if (!other.CompareTag("Player")) return;
            if (_pylonManager.pylonToRemove == this.gameObject)
            {
                _pylonManager.pylonToRemove = null;
            }

            if (_hintText != null) _hintText.text = "";
        }
    }
}

PylonManager.cs (Modified)

This script is simplified. AttemptPylonAction is removed, and we now have a clear RemovePylon method that PlayerInputController will call after a successful hold.


using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Events;

namespace Managers
{
    public class PylonManager : MonoBehaviour
    {
        public List<GameObject> activePylons = new List<GameObject>();
        private HashSet<Vector3> _pylonPositions = new HashSet<Vector3>();

        [SerializeField] private int maxPylons = 3;
        public GameObject pylonPrefab;

        public UnityEvent<Vector3, Vector3, Vector3> onTriangleFormed = new UnityEvent<Vector3, Vector3, Vector3>();
        public UnityEvent<GameObject> onPylonRegistered = new();
        public UnityEvent onPylonsCleared = new UnityEvent();
        public UnityEvent<GameObject> onPylonRemoved = new UnityEvent<GameObject>();

        public GameObject pylonToRemove;
        private AudioSource _audioSource;

        private void Awake()
        {
            _audioSource = GetComponent<AudioSource>();
        }

        // This method is now only for placing pylons.
        public void RegisterPylon(Transform playerTransform)
        {
            if (activePylons.Count >= maxPylons) return;

            Vector3 snappedPos = GetSnappedPosition(playerTransform.position);
            if (_pylonPositions.Contains(snappedPos)) return;

            GameObject newPylon = Instantiate(pylonPrefab, snappedPos, Quaternion.identity);
            if (activePylons.Count == 0)
            {
                newPylon.GetComponent<SpriteRenderer>().color = Color.yellow;
            }

            activePylons.Add(newPylon);
            _pylonPositions.Add(snappedPos);
            Debug.Log($"Pylon {activePylons.Count} placed at {snappedPos}.");
            onPylonRegistered.Invoke(newPylon);
        }

        // NEW: This is now the dedicated method for removal, called by the controller.
        public void RemovePylon(GameObject pylon)
        {
            if (!activePylons.Contains(pylon)) return;
            
            Debug.Log($"Pylon at {pylon.transform.position} removed.");
            activePylons.Remove(pylon);
            _pylonPositions.Remove(pylon.transform.position);
            
            onPylonRemoved.Invoke(pylon); // Notify listeners like the line visualizer.
            
            Destroy(pylon);
        }

        public void OnPylonInteracted(GameObject interactedPylon)
        {
            if (activePylons.Count != maxPylons) return;
            if (interactedPylon == activePylons[0])
            {
                Debug.Log("Triangle formed!");
                onTriangleFormed.Invoke(activePylons[0].transform.position, activePylons[1].transform.position, activePylons[2].transform.position);
                _audioSource.Play();
                ClearPylons();
            }
        }

        public void ClearPylons()
        {
            foreach (var pylon in activePylons) Destroy(pylon.gameObject);
            activePylons.Clear();
            _pylonPositions.Clear();
            onPylonsCleared.Invoke();
        }

        private Vector3 GetSnappedPosition(Vector3 rawPos)
        {
            float snappedX = Mathf.Floor(rawPos.x);
            float snappedY = Mathf.Floor(rawPos.y);
            return new Vector3(snappedX, snappedY, rawPos.z);
        }
    }
}

PlayerInputController.cs (Modified)

This script contains the core logic for differentiating between a "tap" to place and a "hold" to remove.


using Level;
using Managers;
using UnityEngine;
using UnityEngine.InputSystem; // NEW: Required for context access

namespace Player
{
    public class PlayerInputController : MonoBehaviour
    {
        private InputSystem_Actions _playerInput;
        private World _world;
        
        // --- Hold Logic Fields ---
        [Header("Hold to Remove Settings")]
        [SerializeField] private float _holdDuration = 1.2f; // How long to hold for removal.
        private float _holdTimer = 0f;
        private bool _isHoldingForRemoval = false;
        private Pylon _pylonToPotentiallyRemove;
        // -------------------------

        void Awake()
        {
            _playerInput = new InputSystem_Actions();
            _world = World.Instance;

            // Subscribe to the different phases of the input action
            _playerInput.Player.PlacePylon.started += OnPylonActionStarted;
            _playerInput.Player.canceled += OnPylonActionCanceled;
        }

        void OnEnable() { _playerInput.Enable(); }
        void OnDisable() { _playerInput.Disable(); }

        private void OnPylonActionStarted(InputAction.CallbackContext context)
        {
            GameObject markedPylonObject = _world.pylonManager.pylonToRemove;
            
            // Check if player is in range of a removable pylon
            if (markedPylonObject != null)
            {
                // Check the rule: Cannot remove first pylon if 3 are active
                if (_world.pylonManager.activePylons.Count == 3 && _world.pylonManager.activePylons[0] == markedPylonObject)
                {
                    Debug.Log("Cannot remove the first pylon while all three are active.");
                    return; // Do not start hold
                }

                // Start the hold process
                _isHoldingForRemoval = true;
                _holdTimer = 0f;
                _pylonToPotentiallyRemove = markedPylonObject.GetComponent<Pylon>();
            }
            else
            {
                // If no pylon is marked for removal, just place one immediately (tap action)
                _world.pylonManager.RegisterPylon(_world.crewManager.Leader.transform);
            }
        }

        private void OnPylonActionCanceled(InputAction.CallbackContext context)
        {
            // If the button is released, always cancel the hold
            if (_isHoldingForRemoval)
            {
                Debug.Log("Hold canceled.");
                _isHoldingForRemoval = false;
                _holdTimer = 0f;
                if (_pylonToPotentiallyRemove != null)
                {
                    _pylonToPotentiallyRemove.UpdateHoldProgress(0); // Reset UI
                }
                _pylonToPotentiallyRemove = null;
            }
        }

        void Update()
        {
            // --- Movement Logic (unchanged) ---
            Vector2 moveInput = _playerInput.Player.Move.ReadValue<Vector2>();
            if (_world.crewManager.LeaderMovement != null)
            {
                _world.crewManager.LeaderMovement.Move(moveInput);
            }
            // ------------------------------------

            // --- Hold Timer Logic ---
            if (_isHoldingForRemoval)
            {
                _holdTimer += Time.deltaTime;
                float progress = _holdTimer / _holdDuration;
                
                // Update the visual feedback on the pylon
                if (_pylonToPotentiallyRemove != null)
                {
                    _pylonToPotentiallyRemove.UpdateHoldProgress(progress);
                }

                // Check if the hold is complete
                if (_holdTimer >= _holdDuration)
                {
                    Debug.Log("Hold complete! Removing pylon.");
                    _world.pylonManager.RemovePylon(_pylonToPotentiallyRemove.gameObject);
                    
                    // Reset state
                    _isHoldingForRemoval = false;
                    _holdTimer = 0f;
                    _pylonToPotentiallyRemove = null;
                }
            }
            // NEW: If player walks away from pylon while holding, cancel the hold
            else if (_world.pylonManager.pylonToRemove == null && _pylonToPotentiallyRemove != null)
            {
                _pylonToPotentiallyRemove.CancelHold();
                _pylonToPotentiallyRemove = null;
                _isHoldingForRemoval = false;
            }
        }
    }
}
